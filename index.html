def factorial(n):
    if n==1 or n==0:
        return 1
    else:
        return n*factorial(n-1)

def fibonacci(n):
    if n<=1:
        return n
    else:
        return fibonacci(n-1)+fibonacci(n-2)

class doubly:
    def __init__(self, value):
        self.data = value
        self.next = None
        self.prev = None


head = None


def insert():
    global head

    value = int(input("Enter Value : "))
    newnode = doubly(value)

    if head == None:
        head = newnode
    else:
        temp = head

        while temp.next != None:
            temp = temp.next

        temp.next = newnode
        newnode.prev = temp

    print("Node Inserted")


def delete():
    global head

    if head == None:
        print("List Underflow")
    else:
        temp = head
        head = head.next

        if head != None:
            head.prev = None

        temp.next = None

        print("First Node Deleted")


def display():
    global head

    temp = head

    if temp == None:
        print("List Empty")
    else:
        while temp.next != None:
            print(temp.data, end=" <-> ")

stack linked list

class Node:
    def __init__(self, data):
        self.data = data
        self.next = None


class Stack:
    def __init__(self):
        self.top = None

    def push(self, data):
        new_node = Node(data)
        new_node.next = self.top
        self.top = new_node
        print(data, "pushed into stack")

    def pop(self):
        if self.top is None:
            print("Stack Underflow")
            return

        temp = self.top
        self.top = self.top.next
        print("Popped element:", temp.data)

    def peek(self):
        if self.top is None:
            print("Stack is empty")
        else:
            print("Top element:", self.top.data)

    def display(self):
        if self.top is None:
            print("Stack is empty")
            return

        temp = self.top
        print("Stack elements:")

        while temp:
            print(temp.data, end=" ")
            temp = temp.next

        print()


stack = Stack()

while True:
    print("\n1. Push")
    print("2. Pop")
    print("3. Peek")
    print("4. Display")
    print("5. Quit")

    choice = int(input("Enter your choice: "))

    if choice == 1:
        data = int(input("Enter element: "))
        stack.push(data)

    elif choice == 2:
        stack.pop()

    elif choice == 3:
        stack.peek()

    elif choice == 4:
        stack.display()

    elif choice == 5:
        print("Program terminated.")
        break

        print("Invalid choice")
